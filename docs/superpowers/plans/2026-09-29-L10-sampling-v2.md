> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

Branch: feat/l10-sampling-v2

**Goal:** Draw the main breadth sample so that each Jev dollar finds as many firsthand problems as possible while every eligible comment keeps a known, nonzero selection probability, so prevalence estimates stay unbiased.

**Architecture:** New module `src/atlas/sampling/design_v2.py` (and `yield_alloc.py` if it grows) on top of the merged L2 code (`select._draw_rows`, `select._finish`, `select._check`, `select.write_manifest`). Design strata use free code signals only: a frozen pain-word flag, a comment-length bin, a half-year, and a thread-type group. Allocation maximizes expected firsthand problems per token subject to a sampling-rate floor in every stratum. Reuse L2 functions; do not modify `select.py` except to import from it.

```python
# tests/sampling/test_yield_alloc.py
import pytest

from atlas.sampling.yield_alloc import allocate_by_yield


def test_allocation_prefers_high_yield_but_respects_floor_and_budget():
    N = {"pain": 100_000, "nopain": 900_000}
    p = {"pain": 0.30, "nopain": 0.03}        # expected firsthand rate (from the pilot)
    c = {"pain": 500.0, "nopain": 400.0}      # expected tokens per screen call
    alloc = allocate_by_yield(N, p, c, budget_tokens=100_000_000, floor_rate=0.02)
    spent = sum(alloc[h] * c[h] for h in N)
    assert spent <= 100_000_000
    assert alloc["nopain"] >= 0.02 * N["nopain"]          # floor holds
    assert alloc["pain"] / N["pain"] > alloc["nopain"] / N["nopain"]
    assert all(0 < alloc[h] <= N[h] for h in N)


def test_floor_alone_exceeding_budget_raises():
    with pytest.raises(ValueError):
        allocate_by_yield({"a": 10_000_000}, {"a": 0.1}, {"a": 500.0},
                          budget_tokens=1_000_000, floor_rate=0.05)


def test_expected_positives_reported():
    from atlas.sampling.yield_alloc import expected_positives
    assert expected_positives({"a": 100, "b": 10}, {"a": 0.1, "b": 0.5}) == pytest.approx(15)
```
